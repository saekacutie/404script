  <script>
    let qr;
    function initQR(text) {
      document.getElementById('qrcode').innerHTML = '';
      qr = new QRCode(document.getElementById('qrcode'), {
        text: text,
        width: 168,
        height: 168,
        colorDark: '#0a0d0a',
        colorLight: '#eafff0',
        correctLevel: QRCode.CorrectLevel.M
      });
      document.getElementById('qrCaption').textContent = text;
    }

    function updateQR() {
      const val = document.getElementById('qrInput').value.trim() || 'https://example.com';
      initQR(val);
    }

    document.getElementById('qrInput').addEventListener('keydown', function(e) {
      if (e.key === 'Enter') updateQR();
    });

    function setTheme(main, dim, accent) {
      const root = document.documentElement.style;
      root.setProperty('--text', main);
      root.setProperty('--text-dim', dim);
      root.setProperty('--cyan', accent);
    }

    function downloadPNG() {
      html2canvas(document.getElementById('banner'), {
        backgroundColor: '#0a0d0a',
        scale: 2
      }).then(function(canvas) {
        const link = document.createElement('a');
        link.download = 'banner.png';
        link.href = canvas.toDataURL('image/png');
        link.click();
      });
    }

    // Saves the page AS IT CURRENTLY LOOKS - all your typed edits, the
    // theme you picked, the QR content - into a standalone .html file.
    // Open that file later (or re-upload/paste it back to me) and it
    // starts up already showing your edits instead of the defaults.
    function downloadHTML() {
      const clone = document.documentElement.cloneNode(true);
      // Bake the current QR into a plain <img> so the saved file shows
      // the right code immediately, before its own script re-runs.
      const qrEl = clone.querySelector('#qrcode');
      const liveCanvas = document.querySelector('#qrcode canvas');
      if (qrEl && liveCanvas) {
        qrEl.innerHTML = '<img src="' + liveCanvas.toDataURL('image/png') + '" width="168" height="168">';
      }
      const html = '<!DOCTYPE html>\n' + clone.outerHTML;
      const blob = new Blob([html], { type: 'text/html' });
      const link = document.createElement('a');
      link.download = 'banner-edited.html';
      link.href = URL.createObjectURL(blob);
      link.click();
      URL.revokeObjectURL(link.href);
    }

    document.getElementById('dateField').textContent = new Date().toISOString().slice(0, 10);
    initQR('Facebook.com/saekatojirp');
  </script>
